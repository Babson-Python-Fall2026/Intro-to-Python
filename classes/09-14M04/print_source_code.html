import os
import sys
import threading

_write_lock = threading.Lock()  # crude stand-in for real stream locking

def int_to_str(n):
    if n == 0:
        return "0"
    negative = n < 0
    n = abs(n)
    digits = []
    while n > 0:
        digits.append(chr(ord('0') + (n % 10)))
        n //= 10
    if negative:
        digits.append('-')
    return ''.join(reversed(digits))

def float_to_str(f):
    """Convert a float to an EXACT decimal string, without calling
    str()/repr()/format() on the float. Real CPython uses a shortest-
    round-tripping-string algorithm (historically David Gay's dtoa, now
    Grisu/Ryu-style). We can't easily hand-roll THAT, but we can be
    numerically correct by using float.as_integer_ratio(), which exposes
    the EXACT fraction the float represents as arbitrary-precision
    integers -- then do long division by hand, digit by digit."""
    if f != f:                      # NaN check, without math.isnan
        return "nan"
    if f in (float('inf'), float('-inf')):
        return "inf" if f > 0 else "-inf"
    negative = f < 0
    f = abs(f)

    numerator, denominator = f.as_integer_ratio()   # exact fraction, no rounding
    whole = numerator // denominator
    remainder = numerator % denominator

    digits = []
    seen = {}
    for _ in range(50):             # safety cap against infinite repeating decimals
        if remainder == 0:
            break
        if remainder in seen:       # detected a repeat -> stop rather than loop forever
            break
        seen[remainder] = True
        remainder *= 10
        digit = remainder // denominator
        digits.append(chr(ord('0') + digit))
        remainder %= denominator

    frac_str = ''.join(digits) if digits else "0"
    result = int_to_str(whole) + '.' + frac_str
    return '-' + result if negative else result

def value_to_str(value, _seen=None):
    if _seen is None:
        _seen = set()

    if value is None:
        return "None"
    elif isinstance(value, bool):
        return "True" if value else "False"
    elif isinstance(value, int):
        return int_to_str(value)
    elif isinstance(value, float):
        return float_to_str(value)
    elif isinstance(value, str):
        return value
    elif isinstance(value, (list, tuple)):
        if id(value) in _seen:                       # guard against self-referencing containers
            return "[...]" if isinstance(value, list) else "(...)"
        _seen = _seen | {id(value)}
        opener, closer = ('[', ']') if isinstance(value, list) else ('(', ')')
        inner = ', '.join(
            (f"'{v}'" if isinstance(v, str) else value_to_str(v, _seen))
            for v in value
        )
        return opener + inner + closer
    elif isinstance(value, dict):
        if id(value) in _seen:
            return "{...}"
        _seen = _seen | {id(value)}
        inner = ', '.join(f"{value_to_str(k, _seen)}: {value_to_str(v, _seen)}"
                           for k, v in value.items())
        return '{' + inner + '}'
    elif isinstance(value, set):
        if not value:
            return "set()"          # real Python quirk: {} means empty dict, not empty set
        return '{' + ', '.join(value_to_str(v, _seen) for v in value) + '}'
    elif hasattr(value, '__repr__'):
        return value.__repr__()     # can't hand-write a stringifier for every class ever written
    else:
        raise TypeError(f"my_print doesn't know how to stringify {type(value)}")

def _stdout_encoding():
    """Real output encoding depends on locale, platform, and whether the
    destination is a terminal at all -- we ask the OS instead of guessing."""
    return os.device_encoding(1) or sys.getdefaultencoding()

def my_print(*args, sep=' ', end='\n', file=None, flush=False):
    if sep is not None and not isinstance(sep, str):
        raise TypeError(f"sep must be None or a string, not {type(sep).__name__}")
    if end is not None and not isinstance(end, str):
        raise TypeError(f"end must be None or a string, not {type(end).__name__}")
    sep = ' ' if sep is None else sep
    end = '\n' if end is None else end

    text = sep.join(value_to_str(a) for a in args) + end

    if file is None:
        write = lambda s: os.write(1, s.encode(_stdout_encoding()))
        do_flush = (lambda: os.fsync(1)) if flush else (lambda: None)
    else:
        if not hasattr(file, 'write'):
            raise AttributeError(f"{type(file).__name__} object has no attribute 'write'")
        write = file.write
        do_flush = file.flush if (flush and hasattr(file, 'flush')) else (lambda: None)

    with _write_lock:   # so two threads calling my_print don't interleave mid-write
        write(text)
        do_flush()


my_print("Laptop", 999.99, {"tags": ["electronics", "office"]}, {1, 2, 3}, True, None)
